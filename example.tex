  \section{Work experience}

  % You can enable numbering of boxes
  \jobNumberingEnable

  \job{
    title = The granddad,
    start = 1963,
    end = 1966,
    institute = First body,
    where = Somewhere,
  }

  % Changing the color of the frame also changes the color of the title
  \colorlet{jobframe}{BrickRed}
  % Numbering can be disabled, in which case the counter is not incremented
  \jobNumberingDisable
  \job{
    title = The granddad,
    start = 1963,
    end = 1966,
    institute = First body,
    where = Somewhere,
  }

  % You can change how the title color is computed from the color frame
  % This change is not automatically reverted
  \RenewDocumentCommand{\jobTitleColor}{m}{\colorlet{jobtitle}{Blue!60!#1}}
  \job{
    title = The clown,
    start = 1966,
    end = 1969,
    institute = Second body,
    where = Somewhere,
    description = {
      {Stop, you're making me giddy!},
    },
  }

  \jobNumberingEnable
  % You can also change the symbol appearing before the number of the box
  \RenewDocumentCommand{\jobCounterTitle} {} {Work~}

    start = 1970,
    end = 1974,

  \section{Project}

  % Every "description" key (i.e., not just for the project boxes) can accept multiple paragraphs, or uses of environments.
  % Paragraphs are separated by a comma. To avoid any surprise, I highly recommend enclosing each paragraph between { and }

  \project{
    title = Sonic Screwdriver,
    date = 1966, % Not required
    role = Conceptor,
    description = { % Not required
      {It makes noise!},
      { % Start a new paragraph; beware that whitespaces are not ignored after an {
        \begin{itemize}
          \item
          \url{https://github.com/DocSkellington/academic-cv},
          \item
          \url{https://zenodo.org/}
        \end{itemize}
      },
    },
  }
