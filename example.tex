  \section{Work experience}

  % You can enable numbering of boxes
  \jobNumberingEnable

  \job{
    title = The granddad,
    start = 1963,
    end = 1966,
    institute = First body,
    where = Somewhere,
  }
  % Changing the color of the frame also changes the color of the title
  \colorlet{jobframe}{BrickRed}

  % Numbering can be disabled, in which case the counter is not incremented
  \jobNumberingDisable
  \job{
    title = The granddad,
    start = 1963,
    end = 1966,
    institute = First body,
    where = Somewhere,
  }
  \jobNumberingEnable
  % You can also change the letters appearing before the number of the box
  \RenewDocumentCommand{\jobCounterTitle} {} {Work~}
  \job{
    title = The clown,
    start = 1966,
    end = 1969,
    institute = Second body,
    where = Somewhere,
    description = {
      {Stop, you're making me giddy!},
    },
  }
  % You can change how the title color is computed from the color frame
  % This change is not automatically reverted
  \RenewDocumentCommand{\jobTitleColor}{m}{\colorlet{jobtitle}{Blue!60!#1}}
  % The format used to number a box can be completely modified
  \RenewDocumentCommand{\jobCounterFormat}{}{\jobCounterTitle\Roman{\tcbcounter}}

    end = 1974,
